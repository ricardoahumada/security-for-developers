### Password-Based Authentication (Knowledge Factor)

```python
import bcrypt
import secrets
from datetime import datetime, timedelta
from dataclasses import dataclass
from typing import Optional

@dataclass
class User:
    id: str
    email: str
    password_hash: str
    failed_attempts: int = 0
    locked_until: Optional[datetime] = None
    created_at: datetime = None

class AuthenticationService:
    def __init__(self):
        self.max_failed_attempts = 5
        self.lockout_duration = timedelta(minutes=30)
        self.min_password_length = 12
    
    def hash_password(self, password: str) -> str:
        """Hash password using bcrypt with salt"""
        salt = bcrypt.gensalt(rounds=12)
        hashed = bcrypt.hashpw(password.encode('utf-8'), salt)
        return hashed.decode('utf-8')
    
    def verify_password(self, password: str, hashed: str) -> bool:
        """Verify password against hash"""
        return bcrypt.checkpw(
            password.encode('utf-8'), 
            hashed.encode('utf-8')
        )
    
    def validate_password_strength(self, password: str) -> tuple[bool, list[str]]:
        """Validate password meets security requirements"""
        errors = []
        
        if len(password) < self.min_password_length:
            errors.append(f"Password must be at least {self.min_password_length} characters")
        
        if not any(c.isupper() for c in password):
            errors.append("Password must contain uppercase letter")
        
        if not any(c.islower() for c in password):
            errors.append("Password must contain lowercase letter")
        
        if not any(c.isdigit() for c in password):
            errors.append("Password must contain digit")
        
        if not any(c in "!@#$%^&*()_+" for c in password):
            errors.append("Password must contain special character")
        
        return len(errors) == 0, errors

# Usage Example
auth_service = AuthenticationService()

# Password validation
password = "SecureBank2024!"
is_valid, errors = auth_service.validate_password_strength(password)
print(f"Password valid: {is_valid}")
print(f"Errors: {errors}")

# Password hashing
hashed_password = auth_service.hash_password(password)
print(f"Hashed password: {hashed_password}")

# Password verification
is_correct = auth_service.verify_password(password, hashed_password)
print(f"Password verification: {is_correct}")
```

### TOTP Two-Factor Authentication (Possession Factor)

```python
import pyotp
import qrcode
from io import BytesIO
import base64

class TOTPService:
    def __init__(self):
        self.totp = pyotp.TOTP('s3cr3t_k3y_32_bytes_long!')
    
    def generate_secret(self) -> str:
        """Generate a new TOTP secret for user"""
        secret = pyotp.random_base32()
        return secret
    
    def generate_qr_code(self, secret: str, email: str) -> str:
        """Generate QR code for TOTP setup"""
        uri = self.totp.provisioning_uri(
            name=email,
            issuer_name="DataFlow Pro Mobile Banking"
        )
        
        # Generate QR code
        qr = qrcode.QRCode(version=1, box_size=10, border=5)
        qr.add_data(uri)
        qr.make(fit=True)
        
        img = qr.make_image(fill_color="black", back_color="white")
        
        # Convert to base64 for display
        buffer = BytesIO()
        img.save(buffer, format='PNG')
        buffer.seek(0)
        img_base64 = base64.b64encode(buffer.getvalue()).decode()
        
        return img_base64
    
    def verify_totp(self, secret: str, token: str) -> bool:
        """Verify TOTP token"""
        totp = pyotp.TOTP(secret)
        return totp.verify(token, valid_window=1)
```

### Secure Session Implementation

```python
import secrets
import time
from typing import Dict, Optional
from datetime import datetime, timedelta
import json

class SessionManager:
    def __init__(self):
        self.sessions: Dict[str, dict] = {}
        self.session_timeout = timedelta(hours=2)
        self.absolute_timeout = timedelta(hours=24)
        self.refresh_threshold = timedelta(minutes=30)
    
    def create_session(self, user_id: str, additional_data: dict = None) -> str:
        """Create new session"""
        session_id = secrets.token_urlsafe(32)
        
        session_data = {
            'user_id': user_id,
            'created_at': datetime.utcnow(),
            'last_activity': datetime.utcnow(),
            'additional_data': additional_data or {},
            'ip_address': None,  # Set during request
            'user_agent': None   # Set during request
        }
        
        self.sessions[session_id] = session_data
        return session_id
    
    def validate_session(self, session_id: str, 
                        ip_address: str = None, 
                        user_agent: str = None) -> Optional[dict]:
        """Validate session and update activity"""
        if session_id not in self.sessions:
            return None
        
        session = self.sessions[session_id]
        now = datetime.utcnow()
        
        # Check if session has expired
        if now - session['created_at'] > self.absolute_timeout:
            self.delete_session(session_id)
            return None
        
        if now - session['last_activity'] > self.session_timeout:
            self.delete_session(session_id)
            return None
        
        # Update activity and security context
        session['last_activity'] = now
        
        if ip_address and session.get('ip_address'):
            if session['ip_address'] != ip_address:
                # Potential security issue - session hijacking
                self.delete_session(session_id)
                return None
        
        if ip_address:
            session['ip_address'] = ip_address
        
        if user_agent:
            session['user_agent'] = user_agent
        
        return session
    
    def refresh_session(self, session_id: str) -> Optional[str]:
        """Refresh session if within refresh threshold"""
        session = self.sessions.get(session_id)
        if not session:
            return None
        
        now = datetime.utcnow()
        if now - session['last_activity'] < self.refresh_threshold:
            session['last_activity'] = now
            return session_id
        
        return None
    
    def delete_session(self, session_id: str):
        """Delete session"""
        if session_id in self.sessions:
            del self.sessions[session_id]
    
    def get_user_sessions(self, user_id: str) -> list:
        """Get all active sessions for a user"""
        return [
            {'session_id': sid, **session}
            for sid, session in self.sessions.items()
            if session['user_id'] == user_id
        ]

# Usage Example
session_manager = SessionManager()

# Create session
session_id = session_manager.create_session(
    user_id="user123",
    additional_data={'mfa_verified': True, 'risk_score': 0.1}
)

print(f"Session ID: {session_id}")

# Validate session
validated_session = session_manager.validate_session(
    session_id=session_id,
    ip_address="192.168.1.100",
    user_agent="DataFlowMobile/1.0"
)

print(f"Session valid: {validated_session is not None}")
if validated_session:
    print(f"Additional data: {validated_session['additional_data']}")
```

## Example 3: User Registration Flow

```python
import re
from email_validator import validate_email, EmailNotValidError
from typing import Tuple

class UserRegistrationService:
    def __init__(self):
        self.email_validator = validate_email
        self.password_service = AuthenticationService()
    
    def validate_email_address(self, email: str) -> Tuple[bool, str]:
        """Validate email address format and deliverability"""
        try:
            validation = self.email_validator(email, check_deliverability=True)
            return True, validation.normalized
        except EmailNotValidError as e:
            return False, str(e)
    
    def validate_phone_number(self, phone: str) -> Tuple[bool, str]:
        """Validate phone number format"""
        # Simple phone number validation (can be enhanced)
        phone_pattern = r'^\+?1?[-.\s]?\(?([0-9]{3})\)?[-.\s]?([0-9]{3})[-.\s]?([0-9]{4})$'
        
        if re.match(phone_pattern, phone):
            # Normalize phone number
            normalized = re.sub(r'[^\d]', '', phone)
            if len(normalized) == 10:
                normalized = f"+1{normalized}"
            return True, normalized
        
        return False, "Invalid phone number format"
    
    def register_user(self, email: str, password: str, 
                     phone: str, terms_accepted: bool) -> Tuple[bool, dict]:
        """Complete user registration process"""
        validation_results = {
            'email': self.validate_email_address(email),
            'password': self.password_service.validate_password_strength(password),
            'phone': self.validate_phone_number(phone),
            'terms': (terms_accepted, "Terms must be accepted")
        }
        
        # Check if all validations passed
        all_valid = all(result[0] for result in validation_results.values())
        
        if not all_valid:
            errors = {
                field: result[1] 
                for field, result in validation_results.items()
                if not result[0]
            }
            return False, {'errors': errors}
        
        # Create user account
        user_data = {
            'email': validation_results['email'][1],
            'phone': validation_results['phone'][1],
            'password_hash': self.password_service.hash_password(password),
            'mfa_secret': self._generate_mfa_secret(),
            'account_status': 'pending_verification',
            'created_at': datetime.utcnow(),
            'risk_score': 0.0
        }
        
        return True, user_data
    
    def _generate_mfa_secret(self) -> str:
        """Generate MFA secret for new user"""
        import pyotp
        return pyotp.random_base32()

# Usage Example
```

```python
import secrets
import hashlib
from datetime import datetime, timedelta

    def generate_recovery_token(self, user_id: str) -> str:
        """Generate secure recovery token"""
        token = secrets.token_urlsafe(32)
        token_hash = hashlib.sha256(token.encode()).hexdigest()
        
        self.recovery_tokens[token_hash] = {
            'user_id': user_id,
            'created_at': datetime.utcnow(),
            'used': False,
            'attempts': 0
        }

        # Verify user email matches (additional security layer)

        return True, {'message': 'Token validation successful'}

        # Mark token as used
        token_data['used'] = True
        token_data['used_at'] = datetime.utcnow()
        
        return True, {'message': 'Password reset successful'}
```

## Example 5: Authentication Security Monitoring

```python
from enum import Enum
from dataclasses import dataclass
from typing import Dict, List

class RiskLevel(Enum):
    LOW = 0
    MEDIUM = 1
    HIGH = 2
    CRITICAL = 3

@dataclass
class SecurityEvent:
    event_id: str
    user_id: str
    event_type: str
    risk_level: RiskLevel
    details: dict
    timestamp: datetime
    action_taken: str = None

class SecurityMonitoringService:
    def __init__(self):
        self.events: List[SecurityEvent] = []
        self.risk_thresholds = {
            'failed_login_attempts': 3,
            'location_change_radius': 50,  # miles
            'time_of_day_risk_start': 22,  # 10 PM
            'time_of_day_risk_end': 6     # 6 AM
        }
    
    def analyze_login_attempt(self, user_id: str, ip_address: str, 
                            user_agent: str, success: bool, 
                            location: dict = None) -> SecurityEvent:
        """Analyze login attempt for security risks"""
        event_type = "login_success" if success else "login_failure"
        
        # Calculate risk score
        risk_factors = []
        risk_score = 0
        
        # Failed login attempts (would be tracked in real implementation)
        failed_attempts = self._get_failed_attempts(user_id)
        if failed_attempts > self.risk_thresholds['failed_login_attempts']:
            risk_factors.append("multiple_failed_attempts")
            risk_score += 2
        
        # Suspicious location
        if location:
            last_location = self._get_last_location(user_id)
            if last_location:
                distance = self._calculate_distance(
                    last_location, location
                )
                if distance > self.risk_thresholds['location_change_radius']:
                    risk_factors.append("unusual_location")
                    risk_score += 2
        
        # Time-based risk
        current_hour = datetime.utcnow().hour
        if (current_hour >= self.risk_thresholds['time_of_day_risk_start'] or 
            current_hour <= self.risk_thresholds['time_of_day_risk_end']):
            risk_factors.append("off_hours_login")
            risk_score += 1
        
        # New device/browser
        if not self._is_known_device(user_id, user_agent):
            risk_factors.append("new_device")
            risk_score += 1
        
        # Determine risk level
        if risk_score >= 4:
            risk_level = RiskLevel.CRITICAL
        elif risk_score >= 3:
            risk_level = RiskLevel.HIGH
        elif risk_score >= 1:
            risk_level = RiskLevel.MEDIUM
        else:
            risk_level = RiskLevel.LOW
        
        # Create security event
        event = SecurityEvent(
            event_id=secrets.token_urlsafe(16),
            user_id=user_id,
            event_type=event_type,
            risk_level=risk_level,
            details={
                'ip_address': ip_address,
                'user_agent': user_agent,
                'location': location,
                'risk_factors': risk_factors,
                'risk_score': risk_score
            },
            timestamp=datetime.utcnow()
        )
        
        # Take action based on risk level
        event.action_taken = self._determine_action(risk_level, event)
        
        self.events.append(event)
        return event
    
    def _get_failed_attempts(self, user_id: str) -> int:
        """Get failed login attempts count (simplified)"""
        recent_failures = [
            event for event in self.events[-100:]  # Last 100 events
            if (event.user_id == user_id and 
                event.event_type == "login_failure" and
                (datetime.utcnow() - event.timestamp).seconds < 3600)  # Last hour
        ]
        return len(recent_failures)
    
    def _get_last_location(self, user_id: str) -> dict:
        """Get last known location for user"""
        location_events = [
            event for event in self.events
            if (event.user_id == user_id and 
                event.details.get('location'))
        ]
        return location_events[-1].details['location'] if location_events else None
    
    def _is_known_device(self, user_id: str, user_agent: str) -> bool:
        """Check if device is known for user"""
        known_devices = set()
        for event in self.events:
            if event.user_id == user_id and event.event_type == "login_success":
                if event.details.get('risk_score', 0) == 0:  # Successful login with no risk
                    known_devices.add(event.details['user_agent'])
        
        return user_agent in known_devices
    
    def _calculate_distance(self, location1: dict, location2: dict) -> float:
        """Calculate distance between two locations (simplified)"""
        # In real implementation, use proper geolocation calculations
        lat_diff = abs(location1['latitude'] - location2['latitude'])
        lon_diff = abs(location1['longitude'] - location2['longitude'])
        return ((lat_diff ** 2 + lon_diff ** 2) ** 0.5) * 69  # Rough miles
    
    def _determine_action(self, risk_level: RiskLevel, event: SecurityEvent) -> str:
        """Determine action to take based on risk level"""
        if risk_level == RiskLevel.CRITICAL:
            return "BLOCK_LOGIN"
        elif risk_level == RiskLevel.HIGH:
            return "REQUIRE_MFA"
        elif risk_level == RiskLevel.MEDIUM:
            return "LOG_SECURITY_EVENT"
        else:
            return "ALLOW_LOGIN"

    print("---")

print(f"\nTotal security events: {len(monitoring.events)}")
```

## Key Takeaways

1. **Multi-Factor Authentication**: Use multiple factors for enhanced security, especially for financial applications
2. **Session Management**: Implement proper session timeout, refresh mechanisms, and security monitoring
3. **Password Security**: Strong password policies with proper hashing (bcrypt) and validation
4. **Recovery Flows**: Secure token-based password recovery with proper validation and rate limiting
5. **Security Monitoring**: Real-time risk assessment and adaptive security responses

## Best Practices for Mobile Banking

1. **Device Binding**: Implement device fingerprinting and binding
2. **Behavioral Analytics**: Monitor user behavior patterns for anomaly detection
3. **Progressive Security**: Increase security requirements based on transaction risk
4. **Biometric Integration**: Support biometric authentication where available
5. **Continuous Verification**: Implement just-in-time authentication for high-risk actions
